export function sharaf(ctx,tab,control){return{
 async start(row){if(control.blocked)return{blocked:true};if(await tab.url()!=='https://uae.sharafdg.com/')await tab.goto('https://uae.sharafdg.com/');await tab.playwright.domSnapshot();let s=ctx.status.find(x=>x.excel_row===row),box=tab.playwright.getByRole('searchbox',{name:'Submit',exact:true});try{await box.fill(s.keyword)}catch(e){if(!String(e).includes('target token mismatch'))throw e;await tab.playwright.domSnapshot();await tab.playwright.getByRole('searchbox',{name:'Submit',exact:true}).fill(s.keyword)}let value=await tab.playwright.evaluate(()=>document.querySelector('input[type="search"]')?.value);if(value&&value!=='<redacted>')await ctx.verify(row,value);else await fs.appendFile(ctx.dir+'/검색창입력_검증.jsonl',JSON.stringify({row,keyword:s.keyword,input_value:value||'<redacted>',verification:'검색창 fill 인수 J열 원문; 도착 URL 별도 기록',at:new Date().toISOString()})+'\n');await box.press('Enter');await tab.playwright.domSnapshot();s.status='수집 중';await ctx.checkpoint();return{row,submitted:true}},
 async read(){return await tab.playwright.evaluate(()=>{let root=document.querySelector('#hits'),heading=[...document.querySelectorAll('main h3')].map(x=>x.innerText).find(t=>/\([\d,]+\)\s*$/.test(t))||'',m=heading.match(/\(([\d,]+)\)\s*$/),page=Number(document.querySelector('#hits-pagination li.active .go-to-page')?.getAttribute('data-page')||1),next=[...document.querySelectorAll('#hits-pagination li:not(.disabled) .go-to-page')].find(x=>Number(x.getAttribute('data-page'))===page+1);return{url:location.href,page,heading,total_display:m?.[1]||'',total:m?Number(m[1].replace(/,/g,'')):null,next_page:next?page+1:null,cards:[...(root?.querySelectorAll(':scope > .algolia-item')||[])].map((x,i)=>({position:i+1,name:x.querySelector('h4.name')?.innerText.trim()||'',pdp_link:x.querySelector('a.product-link')?.href||'',ad:/Sponsored|Promoted|إعلان/i.test(x.innerText)?'예':'표시 없음'})),head:document.body?.innerText.slice(0,900)||'',main_list:'#hits > .algolia-item; 본 상품 목록만 보존. 필터·중복 제거 없음; promo 주소 인수를 광고 표시로 간주하지 않음'}})},
 async chunk(row){let logs=[],until=Date.now()+13000;while(Date.now()<until){if(control.blocked)return{blocked:true,logs};let d=await this.read(),s=ctx.status.find(x=>x.excel_row===row);if(/보안 확인|Just a moment|captcha|Access Denied|Verify.*human/i.test(d.head)){control.blocked=true;await ctx.blocked(row,d);return{blocked:true,row,logs}}if(!d.cards.length){try{await tab.playwright.locator('#hits > .algolia-item').first().waitFor({state:'visible',timeoutMs:3000})}catch(e){}d=await this.read()}if(new URL(d.url).pathname.startsWith('/product/'))return{unexpected_navigation:true,row,reason:'페이지 넘김 중 상품 상세로 이동; 상세 정보 저장 안 함',logs};let q=new URL(d.url).searchParams.get('q');if(q&&q!==s.keyword)return{yielded:true,reason:'원문 검색 결과 도착 대기',logs};if(!d.cards.length||d.total===null)return{yielded:true,reason:'검색 결과 로딩 대기',logs};let want=Math.min(48,Math.max(0,d.total-(d.page-1)*48));if(d.cards.length!==want||d.cards.some(x=>!x.name||!x.pdp_link))return{yielded:true,reason:'상품 카드 로딩 대기',p:d.page,n:d.cards.length,want,logs};if(!ctx.stored.has(row+':'+d.page)&&d.page!==s.pages+1)return{yielded:true,reason:'앞 페이지부터 재개 필요',p:d.page,want:s.pages+1,logs};await ctx.store(row,{...d,range_display:'화면 페이지 '+d.page+'; 본 목록 '+d.cards.length+'개'},'아랍에미리트 AE#25');logs.push({p:d.page,n:d.cards.length,total:d.total});if(s.cards>=1000&&d.next_page){Object.assign(s,{status:'상한 도달',expected:d.total,reason:'D056 본 목록 순서 상위1,000개',output_cards:1000});await ctx.checkpoint();return{done:true,row,cards:s.cards,expected:d.total,status:s.status,logs}}if(!d.next_page){let e=await ctx.end(row,d,d.total);s.reason='다음 페이지 번호·오른쪽 다음 화살표 없음; 실제'+s.cards+'개·표시'+d.total_display+'개';await ctx.checkpoint();return{...e,logs}}let targetSelector='#hits-pagination li:not(.disabled) .go-to-page[data-page="'+d.next_page+'"]';let pos=await tab.playwright.locator(targetSelector).first().evaluate(x=>{let r=x.getBoundingClientRect();return{top:r.top,left:r.left,bottom:r.bottom,width:r.width,height:r.height,vh:innerHeight}});if(pos.top<60||pos.bottom>pos.vh-50){let wheel=await tab.capabilities.get('cdp');await wheel.send('Input.dispatchMouseEvent',{type:'mouseWheel',x:400,y:400,deltaX:0,deltaY:Math.round(pos.top-pos.vh/2)});await tab.playwright.domSnapshot()}await tab.playwright.locator(targetSelector).first().click();try{await tab.playwright.locator('#hits-pagination li.active .go-to-page[data-page="'+d.next_page+'"]').waitFor({state:'visible',timeoutMs:3000})}catch(e){}await tab.playwright.domSnapshot();let after=await this.read();if(after.page===d.page)return{yielded:true,reason:'페이지 변경 대기',logs}}return{yielded:true,logs}}
}}
